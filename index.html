// Google Apps Script 終極精準同步版 const SHEET_NAME = "工作表1"; // 請確認您的分頁名稱 function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var rows = sheet.getDataRange().getValues(); var attendees = []; // 欄位對應：A:大會編號, B:姓名, C:報到狀態, D:報到時間, E:午餐, F:教育積分, G:資料袋狀態, H:領取人 for (var i = 1; i < rows.length; i++) { var r = rows[i]; if (r[0]) { var bagStatus = String(r[6]).trim(); var isClaimed = (bagStatus === "true" || bagStatus === "已領取"); attendees.push({ badgeNo: String(r[0]).trim(), fullName: String(r[1]).trim(), maskedName: maskName(String(r[1]).trim()), isCheckedIn: String(r[2]) === "true" || String(r[2]) === "已報到", checkinTime: String(r[3]), lunch: String(r[4]), credits: String(r[5]), isBagClaimed: isClaimed, claimedBy: String(r[7] || "") }); } } return ContentService.createTextOutput(JSON.stringify(attendees)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); // 避免多台電腦同時寫入衝突 try { var data = null; if (e.parameter && e.parameter.data) { data = JSON.parse(e.parameter.data); } else if (e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch (pErr) { data = e.parameter; } } else { data = e.parameter; } var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var rows = sheet.getDataRange().getValues(); // 動作一：單筆更新報到狀態 (針對後台點選「取消」或「設為已報到」) if (data && data.action === "updateCheckin") { var targetBadge = String(data.badgeNo).trim(); var isCheckedIn = data.isCheckedIn; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === targetBadge) { sheet.getRange(i + 1, 3).setValue(isCheckedIn ? "已報到" : "未報到"); // C 欄 sheet.getRange(i + 1, 4).setValue(isCheckedIn ? (data.checkinTime || new Date().toString()) : "未報到"); // D 欄 if (data.lunch) sheet.getRange(i + 1, 5).setValue(data.lunch); // E 欄 if (data.credits) sheet.getRange(i + 1, 6).setValue(data.credits); // F 欄 SpreadsheetApp.flush(); // 強制即時同步寫入 break; } } } // 動作二：單筆兌換資料袋 (G欄與H欄) else if (data && data.action === "claimBag") { var targetBadge = String(data.badgeNo).trim(); var claimedBy = String(data.claimedBy).trim(); var isClaimed = data.isBagClaimed !== undefined ? data.isBagClaimed : true; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === targetBadge) { sheet.getRange(i + 1, 7).setValue(isClaimed ? "已領取" : "未領取"); // G 欄 sheet.getRange(i + 1, 8).setValue(isClaimed ? claimedBy : ""); // H 欄 SpreadsheetApp.flush(); break; } } } // 動作三：整張表重置/匯入/刪除同步 (syncAll) else if (data && data.action === "syncAll") { var attendees = typeof data.attendees === "string" ? JSON.parse(data.attendees) : data.attendees; var lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 1, lastRow - 1, 8).clearContent(); } var newRows = []; for (var j = 0; j < attendees.length; j++) { var a = attendees[j]; newRows.push([ a.badgeNo, a.fullName, a.isCheckedIn ? "已報到" : "未報到", a.checkinTime || "", a.lunch || "未填寫", a.credits || "無", a.isBagClaimed ? "已領取" : "未領取", a.claimedBy || "" ]); } if (newRows.length > 0) { sheet.getRange(2, 1, newRows.length, newRows[0].length).setValues(newRows); } SpreadsheetApp.flush(); } lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { if (lock.hasLock()) lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function maskName(name) { if (!name) return ''; var len = name.length; if (len === 2) return name[0] + 'O'; if (len === 3) return name[0] + 'O' + name[2]; if (len === 4) return name[0] + 'OO' + name[3]; if (len > 4) return name[0] + 'O'.repeat(len - 2) + name[len - 1]; return name; }
